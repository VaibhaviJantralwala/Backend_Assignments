Task 2: Find Playlists with More Than 1000 Followers
db.playlists.find(
    { followers: { $gt: 1000 } },
    { _id: 0, name: 1, followers: 1 }
)

Task 3: Find Pop or Hip-Hop Playlists and Sort by Likes
db.playlists.find({
    genre: { $in: ["Pop", "Hip-Hop"] }
}).sort({
    likes: -1
})

Task 4: Find Playlists with Likes Between 500 and 2000
db.playlists.find({
    likes: {
        $gte: 500,
        $lte: 2000
    }
})
.sort({
    followers: -1
})
.limit(3)

Task 5: Find Trending Playlists
